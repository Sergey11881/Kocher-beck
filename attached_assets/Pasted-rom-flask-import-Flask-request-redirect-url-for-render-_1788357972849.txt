def db():
    conn = sqlite3.connect(DB_FILE)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    conn = db()

    conn.execute("""
        CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            order_number TEXT UNIQUE,
            product_type TEXT,
            product_name TEXT,
            client TEXT,
            contact TEXT,
            comment TEXT,
            data TEXT,
            created_at TEXT
        )
    """)

    conn.commit()
    conn.close()


init_db()

@app.route("/create", methods=["POST"])
def create():

    product_type = request.form.get("product_type")

    if product_type not in PRODUCTS:
        flash("Не выбран тип оснастки.")
        return redirect(url_for("index"))

    product = PRODUCTS[product_type]

    data = {}

    for field in product["fields"]:

        name = field[0]
        label = field[1]
        field_type = field[2]

        if field_type == "file":

            file = request.files.get(name)

            if file and file.filename:

                extension = os.path.splitext(file.filename)[1].lower()

                filename = (
                    str(uuid.uuid4())
                    + extension
                )

                filepath = os.path.join(
                    UPLOAD_DIR,
                    filename
                )

                file.save
(filepath)

                data[name] = "/uploads/" + filename

        else:

            value = request.form.get(name, "").strip()

            if value:
                data[name] = value

    import json

    conn = db()

    now = datetime.now
().strftime("%d.%m.%Y %H:%M")

    cursor = conn.execute("""
        INSERT INTO orders
        (
            order_number,
            product_type,
            product_name,
            client,
            contact,
            comment,
            data,
            created_at
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """, (
        "TP-" + datetime.now
().strftime("%Y%m%d") + "-" + str(uuid.uuid4())[:6].upper(),
        product_type,
        product["name"],
        request.form.get("client", ""),
        request.form.get("contact", ""),
        request.form.get("comment", ""),
        json.dumps(data, ensure_ascii=False),
        now
    ))

    order_id = cursor.lastrowid

    conn.commit()
    conn.close()

    flash("Заявка успешно создана.")

    return redirect(
        url_for("order_detail", order_id=order_id)
    )


@app.route("/orders")
def orders():

    conn = db()

    orders = conn.execute("""
        SELECT *
        FROM orders
        ORDER BY id DESC
    """).fetchall()

    conn.close()

    return render_template_string(
        ORDERS,
        orders=orders
    )


@app.route("/order/<int:order_id>")
def order_detail(order_id):

    import json

    conn = db()

    order = conn.execute(
        "SELECT * FROM orders WHERE id = ?",
        (order_id,)
    ).fetchone()

    conn.close()

    if not order:
        return "Заявка не найдена", 404

    data = json.loads(order["data"])

    labels = {}

    for product in PRODUCTS.values():
        for field in product["fields"]:
            labels[field[0]] = field[1]

    return render_template_string(
        DETAIL,
        order=order,
        data=data,
        labels=labels
    )
